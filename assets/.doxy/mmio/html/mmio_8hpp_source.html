45 template <std::uintptr_t Addr, std::size_t BitSize, access_policy AccessPolicy>
46 requires (supported_size<BitSize>)
47 class reg {
48 public:
49 using value_type = size_trait<BitSize>::type;
50 using policy = AccessPolicy;
51
52 static constexpr std::size_t BIT_SIZE = BitSize;
53 static constexpr std::uintptr_t ADDRESS = Addr;
54
55 // Compile-time check to ensure the address is aligned to the register size.
56 // Prevents unaligned access faults on architectures like ARM.
57 static_assert(
58 ADDRESS % (BIT_SIZE / BITS_8) == 0, "MMIO register address is not properly aligned for its data width.");
59
60 private:
65 static volatile value_type& raw() noexcept {
66 return *reinterpret_cast<volatile value_type*>(ADDRESS); // NOLINT(performance-no-int-to-ptr)
67 }

199 template <typename Register, std::size_t Offset, std::size_t Width>
200 requires (Offset + Width <= Register::BIT_SIZE)
201 class field {
202 public:
203 using value_type = Register::value_type;
204 using policy = Register::policy;
205 using register_type = Register;
206
207 static constexpr value_type MASK =
208 Width == Register::BIT_SIZE ? ~value_type{0} : ((value_type(1) << Width) - 1) << Offset;
209
210 field() = delete;
211 field(const field&) = delete;
212 field(field&&) = delete;
213 ~field() = delete;
214 field& operator=(const field&) = delete;
215 field& operator=(field&&) = delete;

217 // ================= READ =================
218
227 [[nodiscard]] static value_type read() noexcept
228 requires (readable<policy>)
229 {
230 return (Register::read() & MASK) >> Offset;
231 }

247 static void write(value_type val) noexcept
248 requires (writable<policy> && readable<policy>)
249 {
250 Register::modify([val](value_type& reg_val) {
251 reg_val &= ~MASK;
252 reg_val |= (val << Offset) & MASK;
253 });
254 }

302 template <typename F>
303 static void modify(F&& func) noexcept(std::is_nothrow_invocable_v<F, value_type&>)
304 requires (writable<policy> && readable<policy> && std::invocable<F, value_type&>)
305 {
306 Register::modify([callable = std::forward<F>(func)](value_type& reg_val) mutable {
307 value_type tmp = (reg_val & MASK) >> Offset;
308 callable(tmp);
309 reg_val &= ~MASK;
310 reg_val |= (tmp << Offset) & MASK;
311 });
312 }

380 }
